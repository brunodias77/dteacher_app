const fetchWithBackoff = async (url, options, retries = 5) => {
  const delays = [1000, 2000, 4000, 8000, 16000];
  for (let i = 0; i < retries; i++) {
    try {
      const response = await fetch(url, options);
      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }
      return response;
    } catch (err) {
      if (i === retries - 1) throw err;
      await new Promise(res => setTimeout(res, delays[i]));
    }
  }
};

const askGemini = async (prompt, schema = null) => {
  const apiKey = ""; 
  const body = { contents: [{ parts: [{ text: prompt }] }] };
  
  if (schema) {
    body.generationConfig = {
      responseMimeType: "application/json",
      responseSchema: schema
    };
  }

  const response = await fetchWithBackoff(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body)
  });

  const data = await response.json();
  const text = data.candidates?.[0]?.content?.parts?.[0]?.text;
  
  if (!text) throw new Error("Sem resposta da IA");
  return schema ? JSON.parse(text) : text;
};

const GeneratorScreen = ({ words, setWords, addToDeck, pickWord, audio }) => {
  const [sentences, setSentences] = useState([]);
  const [loading, setLoading] = useState(false);
  const [addedCards, setAddedCards] = useState(new Set());
  const [errorMsg, setErrorMsg] = useState('');

  const generate = async () => {
    if (!words.trim()) return;
    setLoading(true);
    setErrorMsg('');

    try {
      const prompt = `Você é um professor de inglês experiente ajudando um estudante brasileiro adulto.
Gere até 5 frases naturais em inglês utilizando as seguintes palavras ou expressões: ${words}.
As frases devem ser úteis para o dia a dia.
Retorne APENAS um array de objetos JSON, onde cada objeto contém:
- "english": a frase em inglês.
- "portuguese": a tradução da frase para o Português do Brasil (PT-BR).`;

      const result = await askGemini(prompt, {
        type: "ARRAY",
        items: {
          type: "OBJECT",
          properties: { english: { type: "STRING" }, portuguese: { type: "STRING" } },
          required: ["english", "portuguese"]
        }
      });
      setSentences(result);
    } catch (err) {
      console.error(err);
      setErrorMsg("Desculpe, ocorreu um erro ao gerar as frases. Tente novamente.");
    } finally {
      setLoading(false);
    }
  };

  const handleAddAll = () => {
    sentences.forEach((s, i) => {
      setAddedCards(prev => new Set(prev).add(i));
      addToDeck(s);
    });
  };

  const handleAdd = (item, idx) => {
    setAddedCards(prev => new Set(prev).add(idx));
    addToDeck(item);
  };

  return (
    <div className="animate-fadeSlide space-y-10">
      <div>
        <div className="caps-sm font-mono text-[11px] text-[var(--text-lo)] mb-2">01 · Gerador</div>
        <h1 className="caps-lg text-[38px] font-semibold leading-[1.05] tracking-tight max-w-[28ch]">
          Que palavras quer <span className="font-display italic text-[var(--accent)]">praticar</span> hoje?
        </h1>
        <p className="text-[var(--text-mid)] text-[14px] mt-3 max-w-[52ch]">
          Escreva até cinco palavras. A IA gera frases naturais, traduções e vocabulário — tudo em segundos.
        </p>
      </div>

      <div className="surface p-5 sm:p-6">
        <div className="flex flex-col sm:flex-row gap-3">
          <div className="relative flex-1">
            <input
              className="input pr-28 font-mono text-[14px]"
              placeholder="travel, airport, ticket"

  const currentCard = dueCards[0];

  const normalizeStr = (s) => (s || '').toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[.,!?;:]/g, '').trim();
  const checkAnswer = (input, ans) => {
    if (!input || !ans) return false;
    const ni = normalizeStr(input);
    if (ni === normalizeStr(ans)) return true;
    return ans.split(/,|\/|\bou\b/i).some(p => normalizeStr(p) === ni);
  };

  const grade = (g) => {
    if (!currentCard) return;
    const now = Date.now();
    const base = { again: 60*1000, hard: 10*60*1000, good: 24*60*60*1000, easy: 4*24*60*60*1000 }[g];
    
    updateCard(currentCard.id, { 
      nextReview: now + base, 
      repetitions: (currentCard.repetitions || 0) + 1 
    });
    setIsFlipped(false);
    setUserInput('');
  };

  if (deck.length === 0) {
    return (
      <div className="animate-fadeSlide">

          </h1>

const PhoneticsScreen = ({ audio }) => {

  const generatePhonetics = async () => {

    setLoading(true);
    setErrorMsg('');

    try {
      const prompt = `Você é um professor de inglês ajudando um estudante brasileiro.
Transcreva a seguinte frase em inglês criando uma "pronúncia figurada" que usa as letras e acentos do português brasileiro para simular o som do inglês (ex: 'I love you' -> 'ai lâv iu', 'good morning' -> 'gud mórnin', 'my friend' -> 'mai frênd'). NÃO use símbolos fonéticos complexos (IPA).

- "phonetic": a transcrição completa da frase na pronúncia figurada.
- "translation": a tradução da frase para o Português do Brasil (PT-BR).
- "words": um array de objetos mapeando as palavras mais difíceis ou importantes da frase (máximo de 6). Cada objeto deve ter:
    - "word": a palavra em inglês.

        properties: {
          original: { type: "STRING" },
          phonetic: { type: "STRING" },

          words: {
            type: "ARRAY",
            items: {
              type: "OBJECT",
              properties: {
                word: { type: "STRING" },
                phonetic: { type: "STRING" },
                tip: { type: "STRING" }
              },
              required: ["word", "phonetic", "tip"]
            }
          }
        },
        required: ["original", "phonetic", "translation", "words"]

    } catch (err) {
      console.error(err);
      setErrorMsg("Ocorreu um erro ao processar a transcrição. Tente novamente.");

      </div>

          value={input}

          className="input font-mono text-[13px] resize-none leading-relaxed"
        />
        {errorMsg && (
          <div className="mt-3 text-[13px] text-[var(--danger)] flex items-center gap-2">
            <Icon name="alert" size={14}/> {errorMsg}

          <div className="surface-2 p-6 sm:p-10 relative">

              {storyData.title}
            </h2>

            <div className="space-y-8">
              {storyData.paragraphs.map((p, i) => (
                <div key={i} className="group relative">
                  <div className="text-[18px] sm:text-[20px] text-[var(--text-hi)] leading-relaxed font-medium" style={{ textWrap: "pretty" }}>
                    <ClickableSentence text={p.en} onWord={pickWord} />
                  </div>
                  
                  {revealedPars[i] && (
                    <div className="mt-3 text-[14.5px] text-[var(--text-mid)] italic border-l-2 border-[var(--accent)] pl-4 animate-fadeSlide">
                      {p.pt}
                    </div>
                  )}

                  <div className="mt-4 flex items-center gap-3 opacity-10 sm:opacity-0 group-hover:opacity-100 transition-opacity">
                    <button 
                      onClick={() => audio.playAudio(p.en, `imm-p-${i}`)}
                      className="chip-btn ghost text-[10px] !py-1.5 !px-2"
                    >
                      {audio.loadingId === `imm-p-${i}` ? <Spinner size={12}/> : <Icon name="volume" size={12}/>}
                      {audio.playingId === `imm-p-${i}` ? 'A ler...' : 'Ouvir parágrafo'}
                    </button>
                    <button 
                      onClick={() => toggleReveal(i)}
                      className="chip-btn ghost text-[10px] !py-1.5 !px-2"
                    >
                      <Icon name="eye" size={12}/> {revealedPars[i] ? 'Esconder tradução' : 'Revelar tradução'}
                    </button>
                  </div>
                </div>
              ))}
            </div>
          </div>

          <div className="surface p-6 sm:p-8">
            <SectionHead 
              eyebrow="Reading Comprehension" 
              title="Você compreendeu bem?" 
            />
            <div className="space-y-6">
              {storyData.quiz.map((q, idx) => {
                const isAnswered = quizAnswers[idx] !== undefined;
                const isCorrect = quizAnswers[idx] === q.answer;
                
                return (
                  <div key={idx} className="space-y-3">
                    <div className="text-[15px] font-medium text-[var(--text-hi)] flex gap-2">
                      <span className="text-[var(--text-dim)] font-mono tnum">{idx + 1}.</span> 
                      {q.question}
                    </div>
                    <div className="grid grid-cols-1 sm:grid-cols-3 gap-2">
                      {q.options.map((opt, oIdx) => {
                        const isSelected = quizAnswers[idx] === opt;
                        let btnClass = "surface hover:border-[var(--line-strong)] text-left px-4 py-3 transition-colors text-[13.5px] text-[var(--text-mid)]";
                        
                        if (showResults) {
                          if (opt === q.answer) btnClass = "border-[var(--accent)] bg-[var(--accent-soft)] text-[var(--accent)] text-left px-4 py-3 font-semibold";
                          else if (isSelected) btnClass = "border-[var(--danger)] bg-[rgba(248,113,113,0.05)] text-[var(--danger)] text-left px-4 py-3";
                        } else if (isSelected) {
                          btnClass = "border-[var(--line-strong)] bg-[var(--bg-2)] text-[var(--text-hi)] text-left px-4 py-3 font-medium";
                        }

                        return (
                          <button 
                            key={oIdx} 
                            onClick={() => selectAnswer(idx, opt)}
                            disabled={showResults}
                            className={btnClass}
                          >
                            {opt}
                          </button>
                        );
                      })}
                    </div>
                  </div>
                );
              })}
            </div>

            {!showResults && (
              <div className="mt-8 pt-6 hl-t flex justify-end">
                <button 
                  onClick={() => setShowResults(true)}
                  disabled={Object.keys(quizAnswers).length < storyData.quiz.length}
                  className="btn-primary"
                >
                  <Icon name="check" size={14}/> Corrigir Respostas
                </button>
              </div>
            )}

            {showResults && (
              <div className="mt-8 pt-6 hl-t flex items-center justify-between animate-fadeSlide">
                <div className="text-[14px] text-[var(--text-hi)]">
                  Acertou <strong className="text-[var(--accent)] font-mono text-[16px]">{Object.values(quizAnswers).filter((ans, i) => ans === storyData.quiz[i].answer).length}</strong> de 3!
                </div>
                <button onClick={() => setStoryData(null)} className="btn-ghost">
                  <Icon name="sparkles" size={14}/> Gerar outra história
                </button>
              </div>
            )}
          </div>
        </div>
      )}
    </div>
  );
};

                  {d}
                </button>
              ))}
            </div>
          </div>

          <div>
            <label className="flex items-center justify-between text-[12px] text-[var(--text-mid)] cursor-pointer">
              <span>Streak bar</span>
              <input type="checkbox" checked={tweaks.showStreakBar}

          </div>
        </div>
      )}
    </div>
  );

export default function App() {
  const [user, setUser] = useState(null);

  // Authentication Effect
  useEffect(() => {
    if (!auth) return;
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (error) {
        console.error("Auth error", error);
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  const [tweaks, setTweaks] = useState({
    accent: "lime",
    density: "cozy",
    uppercaseLevel: "labels",
    showStreakBar: true
  });

  const setTweak = (k, v) => setTweaks((prev) => ({ ...prev, [k]: v }));

  useEffect(() => {
    const a = ACCENT_MAP[tweaks.accent] || ACCENT_MAP.lime;
    document.documentElement.style.setProperty('--accent', a.color);
    document.documentElement.style.setProperty('--accent-soft', a.soft);
    document.documentElement.style.setProperty('--accent-border', a.border);
    document.body.setAttribute('data-density', tweaks.density || 'cozy');
    document.body.setAttribute('data-caps', tweaks.uppercaseLevel || 'labels');
    document.body.classList.add('grain');
  }, [tweaks]);

  const [activeTab, setActiveTab] = useState(() => localStorage.getItem('bt_tab') || 'generator');
  useEffect(() => { localStorage.setItem('bt_tab', activeTab); }, [activeTab]);

  const [words, setWords] = useState('');
  const [deck, setDeck] = useState([]);
  const [studiedWords, setStudiedWords] = useState([]);
  const [selectedWord, setSelectedWord] = useState(null);
  const [modalLoading, setModalLoading] = useState(false);

  // LIFTED STATES PARA O TUTOR (Preserva a conversa entre ecrãs)
  const [chatMessages, setChatMessages] = useState([]);
  const [teacherMessages, setTeacherMessages] = useState([]);
  const [chatLevel, setChatLevel] = useState('B1');

  // Firestore Listeners
  useEffect(() => {
    if (!user || !db) return;
    
    const deckRef = collection(db, 'artifacts', appId, 'users', user.uid, 'deck');
    const vocabRef = collection(db, 'artifacts', appId, 'users', user.uid, 'vocabulary');

    const unsubDeck = onSnapshot(deckRef, (snapshot) => {
      const data = [];
      snapshot.forEach(d => data.push({ id: d.id, ...d.data() }));
      setDeck(data);
    }, console.error);

    const unsubVocab = onSnapshot(vocabRef, (snapshot) => {
      const data = [];
      snapshot.forEach(d => data.push({ id: d.id, ...d.data() }));
      data.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
      setStudiedWords(data);
    }, console.error);

    return () => {
      unsubDeck();
      unsubVocab();
    };
  }, [user]);

  const addToDeck = async (s) => {
    if (!user) return;
    const newId = 'c' + Date.now() + Math.random().toString(36).substring(2, 9);
    const card = {
      english: s.english || s.front || s.sentence,
      portuguese: s.portuguese || s.back || s.translation,
      nextReview: Date.now(),
      interval: 0,
      easeFactor: 2.5,
      repetitions: 0,
      createdAt: Date.now()
    };
    try {
      await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'deck', newId), card);
    } catch(e) { console.error("Error adding to deck", e); }
  };

  const updateCard = async (id, updates) => {
    if (!user) return;
    try {
      await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'deck', id), updates, { merge: true });
    } catch(e) { console.error("Error updating card", e); }
  };

  const addWordToVocab = async (wordToFetch) => {
    try {
      const prompt = `Você é um professor de inglês experiente ajudando um estudante brasileiro adulto.
Forneça a tradução e um exemplo de uso para a seguinte palavra ou expressão em inglês: "${wordToFetch.trim()}".
Retorne APENAS um objeto JSON com as seguintes propriedades:
- "word": a palavra original em inglês.
- "translation": a tradução mais comum para Português do Brasil (PT-BR).
- "example": uma frase curta e natural de exemplo em inglês usando a palavra.
- "exampleTranslation": a tradução da frase de exemplo para PT-BR.`;

      const parsed = await askGemini(prompt, {
        type: "OBJECT",
        properties: {
          word: { type: "STRING" },
          translation: { type: "STRING" },
          example: { type: "STRING" },
          exampleTranslation: { type: "STRING" }
        },
        required: ["word", "translation", "example", "exampleTranslation"]
      });

      parsed.createdAt = Date.now();
      
      if (user) {
        const newId = 'v' + Date.now() + Math.random().toString(36).substring(2, 9);
        await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'vocabulary', newId), parsed);
        addToDeck({ english: parsed.word, portuguese: parsed.translation });
      }
    } catch (err) {
      console.error(err);
      const errObj = { word: wordToFetch.trim(), translation: "Erro de conexão", example: "", exampleTranslation: "", createdAt: Date.now() };
      if (user) {
         const newId = 'v' + Date.now() + Math.random().toString(36).substring(2, 9);
         await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'vocabulary', newId), errObj);
         addToDeck({ english: wordToFetch.trim(), portuguese: "Erro de conexão" });
      }
    }
  };

  const [audioPlayingId, setAudioPlayingId] = useState(null);
  const [audioLoadingId, setAudioLoadingId] = useState(null);
  const [speeds, setSpeeds] = useState({});
  const AVAILABLE_SPEEDS = [0.5, 0.75, 1, 1.25, 1.5];

  const changeSpeed = (id, dir) => {
    setSpeeds(prev => {
      const cur = prev[id] || 1;
      const i = AVAILABLE_SPEEDS.indexOf(cur);
      const n = Math.max(0, Math.min(AVAILABLE_SPEEDS.length - 1, i + dir));
      return { ...prev, [id]: AVAILABLE_SPEEDS[n] };
    });
  };

  const playAudio = (text, id) => {
    if (audioLoadingId || audioPlayingId === id) return;
    setAudioLoadingId(id);
    
    setTimeout(() => {
      setAudioLoadingId(null);
      setAudioPlayingId(id);
      
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = 'en-US';
        utterance.rate = speeds[id] || 1; 
        
        utterance.onend = () => setAudioPlayingId(null);
        utterance.onerror = () => setAudioPlayingId(null);
        
        window.speechSynthesis.speak(utterance);
      } else {
        setTimeout(() => setAudioPlayingId(null), 1500);
      }
    }, 300);
  };

  const audioObj = { playingId: audioPlayingId, loadingId: audioLoadingId, speeds, playAudio, changeSpeed };
  const dueCards = deck.filter(c => c.nextReview <= Date.now());
  const stats = { streak: 7, todayReviewed: deck.filter(c=>c.repetitions>0).length, todayDue: dueCards.length };

  const renderTab = () => {
    switch (activeTab) {
      case 'generator':
        return <GeneratorScreen
          words={words}
          setWords={setWords}
          addToDeck={addToDeck}
          pickWord={(w) => setSelectedWord({ word: w })}
          audio={audioObj}
        />;
      case 'flashcards':
        return <FlashcardsScreen
          deck={deck}
          updateCard={updateCard}
          stats={stats}
          audio={audioObj}
          setTab={setActiveTab}
        />;
      case 'vocabulary':
        return <VocabularyScreen
          studiedWords={studiedWords}
          addWordToVocab={addWordToVocab}
          practiceWord={(w) => { setWords(w); setActiveTab('generator'); }}
          simulateChat={() => setActiveTab('tutor')}
        />;
      case 'textStudy':
        return <TextStudyScreen
          addToDeck={addToDeck}
          pickWord={(w) => setSelectedWord({ word: w })}
          startTextChat={() => setActiveTab('tutor')}
          audio={audioObj}
        />;
      case 'tutor':
        return <TutorScreen
          chatMessages={chatMessages} setChatMessages={setChatMessages}
          teacherMessages={teacherMessages} setTeacherMessages={setTeacherMessages}
          chatLevel={chatLevel} setChatLevel={setChatLevel}
        />;
      case 'fillin':
        return <FillInScreen addToDeck={addToDeck} />;
      case 'phonetics':
        return <PhoneticsScreen audio={audioObj} />;
      case 'immersion':
        return <ImmersionScreen pickWord={(w) => setSelectedWord({ word: w })} audio={audioObj} />;
      default: return null;
    }
  };

  const dueCount = dueCards.length;

  return (
    <>
      <style dangerouslySetInnerHTML={{ __html: globalStyles }} />
      <div className="min-h-screen flex flex-col lg:flex-row bg-[var(--bg-0)]">
        <aside className="hidden lg:flex lg:flex-col w-[240px] border-r border-[var(--line)] bg-[var(--bg-0)] sticky top-0 h-screen">
          <div className="px-5 py-5 hl-b">
            <div className="flex items-center gap-3">
              <div className="brand-mark"><i/><i/><i/><i/></div>
              <div>
                <div className="text-[15px] font-semibold tracking-tight text-[var(--text-hi)]">btree_english</div>
                <div className="caps-sm font-mono text-[10px] text-[var(--text-lo)]">AI · SRS · v2.0</div>
              </div>
            </div>
          </div>

          {tweaks.showStreakBar && (
            <div className="px-5 py-4 hl-b">
              <div className="flex items-center justify-between mb-2">
                <div className="caps-sm font-mono text-[10px] text-[var(--text-lo)]">Dias Seguidos</div>
                <div className="flex items-center gap-1.5 text-[var(--accent)]">
                  <Icon name="flame" size={12}/>
                  <span className="font-mono text-[13px] font-bold tnum">{stats.streak}</span>
                </div>
              </div>
              <div className="flex gap-1">
                {Array.from({length: 7}).map((_, i) => {
                  const activeBars = stats.streak === 0 ? 0 : ((stats.streak - 1) % 7) + 1;
                  return <div key={i} className={`flex-1 h-1.5 ${i < activeBars ? "bg-[var(--accent)]" : "bg-[var(--line)]"}`}/>;
                })}
              </div>
              <div className="flex justify-between mt-2 text-[10px] font-mono text-[var(--text-dim)] caps-sm">
                <span>S T Q Q S S D</span>
                <span className="tnum">{stats.todayReviewed} hoje</span>
              </div>
            </div>
          )}

          <nav className="flex-1 p-3 space-y-0.5 overflow-y-auto">
            {TABS.map(t => {
              const active = activeTab === t.id;
              const badge = t.id === 'flashcards' && dueCount > 0 ? dueCount : null;
              return (
                <button key={t.id} onClick={() => setActiveTab(t.id)}
                  className={`w-full flex items-center gap-3 px-3 py-2.5 text-left transition-colors relative group ${
                    active ? "bg-[var(--bg-2)] text-[var(--text-hi)]" : "text-[var(--text-mid)] hover:text-[var(--text-hi)] hover:bg-[var(--bg-1)]"
                  }`}>
                  {active && <span className="absolute left-0 top-2 bottom-2 w-[2px] bg-[var(--accent)]"/>}
                  <span className="font-mono text-[10px] text-[var(--text-dim)] tnum w-5">{t.num}</span>
                  <Icon name={t.icon} size={15}/>
                  <span className="text-[13px] font-medium flex-1">{t.label}</span>
                  {badge && <span className="tag accent !py-0.5 !px-1.5 !text-[9px]">{badge}</span>}
                </button>
              );
            })}
          </nav>

          <div className="p-3 hl-t space-y-2">
            <button className="w-full flex items-center justify-between px-3 py-2 text-[12px] text-[var(--text-mid)] hover:text-[var(--text-hi)] hover:bg-[var(--bg-1)]">
              <span className="flex items-center gap-2"><Icon name="command" size={12}/> Atalhos</span>
              <KeyHint keys={['⌘','K']}/>
            </button>
            <div className="px-3 pt-2 hl-t">
              <div className="caps-sm font-mono text-[10px] text-[var(--text-dim)]">
                <span className="inline-block w-1.5 h-1.5 bg-[var(--accent)] mr-1.5 pulse-dot align-middle"/>
                Gemini 2.5 online
              </div>
            </div>
          </div>
        </aside>

        <header className="lg:hidden flex items-center justify-between px-4 py-3 hl-b sticky top-0 bg-[var(--bg-0)] z-20">
          <div className="flex items-center gap-2">
            <div className="brand-mark"><i/><i/><i/><i/></div>
            <span className="font-semibold tracking-tight">btree_english</span>
          </div>
          <div className="flex items-center gap-1.5 text-[var(--accent)]">
            <Icon name="flame" size={12}/>
            <span className="font-mono text-[12px] font-bold tnum">{stats.streak}</span>
          </div>
        </header>

        <main className="flex-1 min-w-0 px-4 sm:px-8 lg:px-12 py-6 lg:py-10 pb-24 lg:pb-10 max-w-[1080px] mx-auto w-full">
          {renderTab()}
        </main>

        <nav className="lg:hidden fixed bottom-0 inset-x-0 bg-[var(--bg-0)] hl-t flex overflow-x-auto z-30">
          {TABS.map(t => {
            const active = activeTab === t.id;
            return (
              <button key={t.id} onClick={() => setActiveTab(t.id)}
                className={`flex-shrink-0 flex flex-col items-center justify-center gap-1 px-4 py-2.5 min-w-[72px] relative ${active ? "text-[var(--accent)]" : "text-[var(--text-lo)]"}`}>
                <Icon name={t.icon} size={16}/>
                <span className="text-[9px] font-mono uppercase tracking-wider">{t.label}</span>
                {active && <span className="absolute top-0 w-6 h-[2px] bg-[var(--accent)]"/>}
              </button>
            );
          })}
        </nav>

        {selectedWord && (
          <div className="fixed inset-0 bg-black/80 backdrop-blur-sm z-40 flex items-center justify-center p-4" onClick={() => setSelectedWord(null)}>
            <div className="surface-2 p-6 max-w-sm w-full" onClick={(e) => e.stopPropagation()}>
              <div className="caps-sm font-mono text-[11px] text-[var(--accent)] mb-3">Nova palavra</div>
              <div className="text-[24px] font-display text-[var(--text-hi)] mb-2">"{selectedWord.word}"</div>
              <div className="text-[13px] text-[var(--text-mid)] mb-6">Adicionar ao vocabulário e criar flashcard?</div>
              <div className="flex gap-2">
                <button onClick={() => setSelectedWord(null)} className="btn-ghost flex-1 justify-center">Cancelar</button>
                <button 
                  onClick={async () => {
                    setModalLoading(true);
                    await addWordToVocab(selectedWord.word);
                    setModalLoading(false);
                    setSelectedWord(null);
                  }} 
                  disabled={modalLoading}
                  className="btn-primary flex-1 justify-center"
                >
                  {modalLoading ? <Spinner size={14}/> : <Icon name="plus" size={14}/>} Confirmar
                </button>
              </div>
            </div>
          </div>
        )}

        <TweaksPanel tweaks={tweaks} setTweak={setTweak}/>
      </div>
    </>
  );
}
